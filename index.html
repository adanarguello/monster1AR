    <a-scene mindar-image="imageTargetSrc: ./targets.mind;" vr-mode-ui="enabled: false" device-orientation-permission-ui="enabled: false">

      <a-assets>

        <video id="vid" src="./animacion.mp4" webkit-playsinline playsinline muted loop crossorigin="anonymous"></video>

      </a-assets>


      <a-camera position="0 0 0" look-controls="enabled: false"></a-camera>


      <a-entity id="target" mindar-image-target="targetIndex: 0">

        <a-video src="#vid" position="0 0 0" width="1" height=“1.95 rotation="0 0 0"></a-video>

      </a-entity>

    </a-scene>


    <script>

      const video = document.querySelector('#vid');

      const btn = document.querySelector('#btn-iniciar');

      const target = document.querySelector('#target');


      // 1. Al tocar el botón, activamos el sonido y preparamos el video

      btn.addEventListener('click', () => {

        video.muted = false; // Devuelve el sonido original al video

        video.play();

        video.pause(); // Lo pausamos esperando a que la cámara vea la imagen

        btn.style.display = 'none'; // Ocultamos el botón

      });


      // 2. Cuando la cámara detecta al monstruo

      target.addEventListener('targetFound', () => {

        video.play();

      });


      // 3. Cuando alejas la cámara

      target.addEventListener('targetLost', () => {

        video.pause();

      });

    </script>
